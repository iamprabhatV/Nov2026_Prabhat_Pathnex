# Jenkinsfile with git integration and Testing

pipeline {
  agent any
  stages {
    stage('Clone Repository') {
      steps {
        git 'https://github.com/iamprabhatV/Travel-site'
      }
    }

    stage('Build') {
      steps {
        echo 'Building project from git repo'
      }
    }

    stage('Test') {
      steps {
        echo 'Running test'
      }
    }

    stage('Deploy') {
      steps {
        echo 'Deploy application'
      }
    }
  }
}
