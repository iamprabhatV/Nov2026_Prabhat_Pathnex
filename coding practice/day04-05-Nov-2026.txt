# Jenkins File - Basic CI Pipeline

pipeline {
  agent any
  stages {
    stage('Build') {
      steps {
        echo 'Building Project'
      }
    }
    stage('Test') {
      steps {
        echo 'Running Tests'
      }
    }
    stage('Deploy') {
      steps{
        echo 'Deploying Application'
      }
    }
  }
}


# GitLab CI/CD - Basic Pipeline

stages:
  - build
  - test
  - deploy

build: 
  stage: build
  script:
    - echo "Building Project"

test: 
  stage: test
  script:
    - echo "Running Tests"

deploy: 
  stage: deploy
  script:
    - echo "Deploying Project"


#Ansible - install and configure Nginx

- name: install and configure Nginx
  hosts: all
  become: yes
  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present
    - name: Start nginx service
      service:
        name: nginx
        state: started
        enabled: yes
      


#Terraform - EC2 with Security Group

resource "aws_security_group" "sg1" {
  name = "sg1_security_group"
  description = "Allow SSH and HTTP"
  ingress {
    from_port = 22
    to_port = 22
    protocol = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    from_port = 80
    to_port = 80
    protocol = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    from_port = 0
    to_port = 0
    protocol = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}
